######## PART 12 script (prod_frontend_check.py, plain public GETs of https://prooflab.co.in):
"""PART 12 - READ-ONLY inspection of the DEPLOYED production frontend (https://prooflab.co.in).
Plain GET requests for public files only (the same files any visitor's browser downloads).
Prints: which JS bundles exist, which backend URLs are compiled in, and whether the Step 6
async (queued) recording code is compiled in / switched on. API keys are redacted."""
import re, sys, urllib.request

BASE = "https://prooflab.co.in"
def get(url):
    with urllib.request.urlopen(urllib.request.Request(url, headers={"User-Agent": "prooflab-readonly-audit"}), timeout=60) as r:
        return r.read().decode("utf-8", errors="replace"), dict(r.headers)

html, headers = get(BASE + "/")
print("index.html bytes:", len(html), "| last-modified:", headers.get("Last-Modified"), "| etag:", headers.get("ETag"))
entry = re.findall(r'src="(/assets/[^"]+\.js)"', html)
print("entry scripts:", entry)
seen, queue, bundles = set(), list(entry), {}
while queue:
    p = queue.pop()
    if p in seen: continue
    seen.add(p)
    try:
        js, _ = get(BASE + p)
    except Exception as e:
        print("fetch failed", p, e); continue
    bundles[p] = js
    for dep in re.findall(r'["\'`](?:\./|/assets/)([A-Za-z0-9_.-]+\.js)["\'`]', js):
        queue.append("/assets/" + dep)
print("JS files fetched:", len(bundles))

alljs = "\n".join(bundles.values())
urls = sorted(set(re.findall(r'https://[a-z0-9.-]+\.(?:run\.app|googleapis\.com|web\.app|firebaseapp\.com)[^"\'`\s)]*', alljs)))
print("\nbackend URLs compiled into the deployed bundle:")
for u in urls:
    print("  ", re.sub(r"key=[A-Za-z0-9_-]+", "key=<redacted>", u))
print("\nstaging URLs present in production bundle:", [u for u in urls if "staging" in u])

markers = {
    "transcription-enqueue (async enqueue call)": "transcription-enqueue",
    "voice-score (sync scoring call)": "voice-score",
    "pl.voiceJob.v3: (round-4+ recovery records)": "pl.voiceJob.v3:",
    "pl.voiceJob. (any recovery marker, round 1+)": "pl.voiceJob.",
    "'Queued — this can take a moment' (async UI text)": "Queued — this can take a moment",
    "'Uploading your recording' (async UI text)": "Uploading your recording",
    "'Saving your explanation' (sync UI text)": "Saving your explanation",
    "'Recordings you kept aside' (round-4+ UI)": "Recordings you kept aside",
    "'Show hidden recordings' (round-7 UI)": "hidden recordings",
    "/transcribe (browser transcriber call, sync path)": "/transcribe",
}
print("\nStep 6 markers in the deployed bundle:")
for label, needle in markers.items():
    files = [p for p, js in bundles.items() if needle in js]
    print(f"  {'YES' if files else 'no '}  {label}  {files[:2]}")

# the async flag: Vite replaces import.meta.env.VITE_ASYNC_TRANSCRIPTION with a literal at build time
for p, js in bundles.items():
    for m in re.finditer(r'.{60}VITE_ASYNC_TRANSCRIPTION.{60}', js):
        print("  flag reference (unreplaced):", p, m.group(0))
print("\nnote: if VITE_ASYNC_TRANSCRIPTION was unset at build time, Vite compiles the check to a constant false")
